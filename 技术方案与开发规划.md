  7. 集成到现有系统

  // 在现有渲染器中集成
  class EnhancedRenderer {
      constructor() {
          this.materialTemplateManager = new MaterialTemplateManager();
          this.templateSelector = new TemplateSelector(this.materialTemplateManager);
      }

      // 渲染时应用材质模板
      renderWithTemplate(object, objectType, objectId) {
          const material = this.materialTemplateManager.getMaterial(
              objectType,
              objectId,
              object.userData
          );

          object.material = material;
          return object;
      }

      // 监听模板变化
      onTemplateChange(callback) {
          this.materialTemplateManager.addObserver(callback);
      }
  }

  9. 关键考虑点

  - 性能优化：材质缓存，避免重复创建
